1.
SELECT id,
		YEAR(date) AS year,
        MONTH(date) AS month,
        DAY(date) AS day,
        date
FROM orders;

2.
SELECT id,
        date,
        DATE_ADD(date, INTERVAL 1 DAY) AS new_date
FROM orders;

3.
SELECT id,
       date,
       UNIX_TIMESTAMP(date) AS seconds_from_start
FROM orders;

4.
SELECT COUNT(*) AS qty_per_period
FROM orders
WHERE date BETWEEN '1996-07-10 00:00:00' AND '1996-10-08 00:00:00';

5.
SELECT id,
	date,
	JSON_OBJECT('id', id, 'date', date) AS json_result
FROM orders;
